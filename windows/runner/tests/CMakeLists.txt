cmake_minimum_required(VERSION 3.15)
project(vertree_shell_icon_tests LANGUAGES CXX RC)
add_executable(vertree_shell_icon_tests shell_icons_test.cpp ../shell_icons.cpp test_icons.rc)
target_compile_features(vertree_shell_icon_tests PRIVATE cxx_std_17)
target_compile_definitions(vertree_shell_icon_tests PRIVATE NOMINMAX UNICODE _UNICODE)
target_compile_options(vertree_shell_icon_tests PRIVATE $<$<COMPILE_LANGUAGE:CXX>:/W4 /WX /utf-8>)
target_link_libraries(vertree_shell_icon_tests PRIVATE advapi32 user32 gdi32 propsys shell32 ole32 uuid)
enable_testing()
add_test(NAME shell_icons COMMAND vertree_shell_icon_tests)
add_executable(vertree_shell_shortcut_tests shell_shortcuts_test.cpp ../shell_shortcuts.cpp)
target_compile_features(vertree_shell_shortcut_tests PRIVATE cxx_std_17)
target_compile_definitions(vertree_shell_shortcut_tests PRIVATE NOMINMAX UNICODE _UNICODE)
target_compile_options(vertree_shell_shortcut_tests PRIVATE /W4 /WX /utf-8)
target_link_libraries(vertree_shell_shortcut_tests PRIVATE propsys shell32 ole32 uuid)
add_test(NAME shell_shortcuts COMMAND vertree_shell_shortcut_tests)
