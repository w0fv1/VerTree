cmake_minimum_required(VERSION 3.21)
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
  project(vertree_file_tools LANGUAGES CXX)
endif()
add_library(vertree_file_tools_core STATIC
  common.cpp path_guard.cpp delete_engine.cpp file_lock_scanner.cpp process_actions.cpp)
target_compile_features(vertree_file_tools_core PUBLIC cxx_std_17)
target_compile_definitions(vertree_file_tools_core PUBLIC
  UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX _WIN32_WINNT=0x0A00)
target_include_directories(vertree_file_tools_core PUBLIC ${CMAKE_CURRENT_SOURCE_DIR})
if(MSVC)
  target_compile_options(vertree_file_tools_core PUBLIC /W4 /EHsc /utf-8)
endif()
target_link_libraries(vertree_file_tools_core PUBLIC advapi32 shell32 psapi bcrypt user32 ole32)
add_executable(vertree_file_worker main.cpp elevation_broker.cpp)
target_link_libraries(vertree_file_worker PRIVATE vertree_file_tools_core)
if(DEFINED BINARY_NAME AND TARGET ${BINARY_NAME})
  add_dependencies(${BINARY_NAME} vertree_file_worker)
  install(TARGETS vertree_file_worker RUNTIME DESTINATION "${CMAKE_INSTALL_PREFIX}" COMPONENT Runtime)
  install(FILES third_party/nlohmann/LICENSE.MIT
    DESTINATION "${CMAKE_INSTALL_PREFIX}/data/licenses/nlohmann-json" COMPONENT Runtime)
  install(FILES third_party/README.md
    DESTINATION "${CMAKE_INSTALL_PREFIX}/data/licenses/file-tools" COMPONENT Runtime)
endif()
option(VERTREE_FILE_TOOLS_TESTS "Build native file-tools tests" ON)
if(VERTREE_FILE_TOOLS_TESTS)
  enable_testing()
  add_executable(vertree_file_lock_holder tests/file_lock_holder.cpp)
  target_link_libraries(vertree_file_lock_holder PRIVATE vertree_file_tools_core)
  add_executable(vertree_file_tools_tests tests/native_tests.cpp)
  target_link_libraries(vertree_file_tools_tests PRIVATE vertree_file_tools_core)
  add_test(NAME vertree_file_tools_native COMMAND vertree_file_tools_tests)
endif()
